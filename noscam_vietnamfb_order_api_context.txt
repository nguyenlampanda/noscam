18:     protected function endpoint(): string

20:         $url = rtrim($this->apiUrl, '/');
21: 
22:         if (!str_ends_with($url, '/api')) {
23:             $url .= '/api';

29:     protected function get(
30:         string $mc,
31:         string $site,
32:         array $params = []
33:     ): array {
34:         $response = Http::acceptJson()
35:             ->timeout(30)
36:             ->retry(2, 500)
37:             ->get(
38:                 $this->endpoint(),
39:                 array_merge([
40:                     'mc' => $mc,
41:                     'site' => $site,
42:                     'api_key' => $this->apiKey,
43:                 ], $params)

51:         string $site,

54:         $response = Http::asForm()
55:             ->acceptJson()
56:             ->timeout(30)
57:             ->retry(2, 500)
58:             ->post(
59:                 $this->endpoint().'?'.http_build_query([
60:                     'mc' => $mc,
61:                     'site' => $site,
62:                     'api_key' => $this->apiKey,
63:                 ]),
64:                 $data
65:             );
66: 
67:         return $this->parseResponse($response);
68:     }

70:     protected function parseResponse($response): array
71:     {
72:         if (!$response->successful()) {
73:             throw new RuntimeException(
74:                 'VietnamFB HTTP error: '.$response->status()
75:             );
76:         }
77: 
78:         $body = trim($response->body());
79:         $lower = strtolower($body);
80: 
81:         if (
82:             str_contains($lower, '<html') ||
83:             str_contains($lower, '<script')
84:         ) {
85:             throw new RuntimeException(
86:                 'VietnamFB trả về HTML/login page.'
87:             );
88:         }
89: 
90:         $json = $response->json();
91: 
92:         if (!is_array($json)) {
93:             throw new RuntimeException(
94:                 'VietnamFB response không hợp lệ.'
95:             );
96:         }
97: 
98:         if (
99:             isset($json['success']) &&
100:             (int) $json['success'] === 0
101:         ) {
102:             throw new RuntimeException(
103:                 (string) (
104:                     $json['message']
105:                     ?? 'VietnamFB API báo lỗi.'
106:                 )
107:             );
108:         }
109: 
110:         return $json;
111:     }

121:     public function services(): array
122:     {
123:         $username = trim(
124:             (string) $this->username
125:         );
126: 
127:         if ($username === '') {
128:             throw new RuntimeException(
129:                 'VietnamFB chưa cấu hình username.'
130:             );
131:         }
132: 
133:         $result = $this->get(
134:             'channel',
135:             'get_channel_info',
136:             [
137:                 'username' => $username,
138:             ]
139:         );
140: 
141:         $services = [];
142: 
143:         foreach ($result as $group => $channels) {
144:             if (!is_array($channels)) {
145:                 continue;
146:             }
147: 
148:             foreach ($channels as $channel) {
149:                 if (!is_array($channel)) {
150:                     continue;
151:                 }
152: 
153:                 $channelNumber =
154:                     $channel['channel']
155:                     ?? $channel['id']
156:                     ?? null;
157: 
158:                 if ($channelNumber === null) {
159:                     continue;
160:                 }
161: 
162:                 $groupName = strtoupper(
163:                     (string) $group
164:                 );
165: 
166:                 $channelId = (string) $channelNumber;
167: 
168:                 $statusText = (string) (
169:                     $channel['status_text']
170:                     ?? ''
171:                 );
172: 
173:                 $services[] = [
174:                     'service' =>
175:                         $groupName.':'.$channelId,
176: 
177:                     'name' =>
178:                         $this->serviceName(
179:                             $groupName,
180:                             $channelId
181:                         ),
182: 
183:                     'rate' =>
184:                         is_numeric(
185:                             $channel['price'] ?? null
186:                         )
187:                             ? (float) $channel['price']
188:                             : 0,
189: 
190:                     'min' =>
191:                         is_numeric($channel['min'] ?? null)
192:                         && (int) $channel['min'] > 0
193:                             ? (int) $channel['min']
194:                             : 1,
195: 
196:                     'max' =>
197:                         is_numeric($channel['max'] ?? null)
198:                         && (int) $channel['max'] > 0
199:                             ? (int) $channel['max']
200:                             : 1000000,
201: 
202:                     'min_verified' =>
203:                         is_numeric($channel['min'] ?? null)
204:                         && (int) $channel['min'] > 0,
205: 
206:                     'max_verified' =>
207:                         is_numeric($channel['max'] ?? null)
208:                         && (int) $channel['max'] > 0,
209: 
210:                     'group' => $groupName,
211: 
212:                     'channel' => $channelId,
213: 
214:                     'status_text' => $statusText,
215: 
216:                     'provider_active' =>
217:                         $this->isChannelActive(
218:                             $channel
219:                         ),
220: 
221:                     'provider_data' => $channel,
222:                 ];
223:             }
224:         }
225: 
226:         return $services;
227:     }

231:         string $channel
232:     ): string {
233:         $label = str_replace(
234:             '_',
235:             ' ',
236:             $group

239:         return trim(
240:             $label.' - Kênh '.$channel

244:     protected function isChannelActive(
245:         array $channel
246:     ): bool {
247:         $status = strtolower(
248:             trim(
249:                 (string) (
250:                     $channel['status_text']
251:                     ?? $channel['status']
252:                     ?? ''
253:                 )
254:             )
255:         );
256: 
257:         if ($status === '') {
258:             return true;
259:         }
260: 
261:         foreach ([
262:             'tắt',
263:             'tat',
264:             'ngưng',
265:             'ngung',
266:             'bảo trì',
267:             'bao tri',
268:             'off',
269:             'inactive',
270:             'disabled',
271:         ] as $word) {
272:             if (str_contains($status, $word)) {
273:                 return false;
274:             }
275:         }
276: 
277:         return true;
278:     }
