   6: async function request(
   7:   endpoint,
   8:   {
   9:     method = 'GET',
  10:     body,
  11:     headers = {},
  12:     signal,
  13:   } = {},
  14: ) {
  15:   const url =
  16:     `${API_BASE_URL}${endpoint}`
  17: 
  18:   const isFormData =
  19:     typeof FormData !==
  20:       'undefined' &&
  21:     body instanceof FormData
  22: 
  23:   const config = {
  24:     method,
  25: 
  26:     headers: {
  27:       Accept:
  28:         'application/json',
  29: 
  30:       ...headers,
  31:     },
  32: 
  33:     signal,
  34:   }
  35: 
  36:   if (body !== undefined) {
  37:     if (isFormData) {
  38:       config.body = body
  39:     } else {
  40:       config.headers[
  41:         'Content-Type'
  42:       ] = 'application/json'
  43: 
  44:       config.body =
  45:         JSON.stringify(body)
  46:     }
  47:   }
  48: 
  49:   let response
  50: 
  51:   try {
  52:     response =
  53:       await fetch(
  54:         url,
  55:         config,
  56:       )
  57:   } catch (error) {
  58:     if (
  59:       error.name ===
  60:       'AbortError'
  61:     ) {
  62:       throw error
  63:     }
  64: 
  65:     throw new Error(
  66:       'Không thể kết nối tới máy chủ. Vui lòng kiểm tra kết nối và thử lại.',
  67:     )
  68:   }
  69: 
  70:   let data = null
  71: 
  72:   const contentType =
  73:     response.headers.get(
  74:       'content-type',
  75:     )
  76: 
  77:   if (
  78:     contentType?.includes(
  79:       'application/json',
  80:     )
  81:   ) {
  82:     try {
  83:       data =
  84:         await response.json()
  85:     } catch {
  86:       data = null
  87:     }
  88:   }
  89: 
  90:   if (!response.ok) {
  91:     let message =
  92:       data?.message ||
  93:       `Yêu cầu thất bại với mã ${response.status}.`
  94: 
  95:     if (response.status === 429) {
  96:       message =
  97:         'Bạn đang gửi quá nhiều yêu cầu. Vui lòng chờ một lúc rồi thử lại.'
  98:     }
  99: 
 100:     if (data?.errors) {
 101:       const validationMessages =
 102:         Object.values(
 103:           data.errors,
 104:         )
 105:           .flat()
 106:           .filter(Boolean)
 107: 
 108:       if (
 109:         validationMessages.length >
 110:         0
 111:       ) {
 112:         message =
 113:           validationMessages.join(
 114:             ' ',
 115:           )
 116:       }
 117:     }
 118: 
 119:     const error =
 120:       new Error(message)
 121: 
 122:     error.status =
 123:       response.status
 124: 
 125:     error.data =
 126:       data
 127: 
 128:     throw error
 129:   }
 130: 
 131:   return data
 132: }
